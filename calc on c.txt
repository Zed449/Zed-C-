#include <stdio.h>

// Function declarations
void add(float num1, float num2);
void subtract(float num1, float num2);
void multiply(float num1, float num2);
void divide(float num1, float num2);
void handle_error();

int main() {
    float num1, num2;
    char operatorSymbol;

    // Taking user input
    printf("Enter first number: ");
    scanf("%f", &num1);

    printf("Enter an operator (+, -, *, /): ");
    scanf(" %c", &operatorSymbol); // Space before %c to handle newline issues

    printf("Enter second number: ");
    scanf("%f", &num2);

    // Processing input
    if (operatorSymbol == '+') {
        add(num1, num2);
    }
    else if (operatorSymbol == '-') {
        subtract(num1, num2);
    }
    else if (operatorSymbol == '*') {
        multiply(num1, num2);
    }
    else if (operatorSymbol == '/') {
        if (num2 == 0) {
            handle_error();
        } else {
            divide(num1, num2);
        }
    } 
    else {
        handle_error();
    }

    return 0;
}

// Function definitions
void add(float num1, float num2) {
    printf("Result: %.2f\n", num1 + num2);
}

void subtract(float num1, float num2) {
    printf("Result: %.2f\n", num1 - num2);
}

void multiply(float num1, float num2) {
    printf("Result: %.2f\n", num1 * num2);
}

void divide(float num1, float num2) {
    printf("Result: %.2f\n", num1 / num2);
}

void handle_error() {
    printf("Error: Invalid operator or division by zero.\n");
}
